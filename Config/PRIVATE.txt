# ============ EasyTier 核心 ============
# luci-app-easytier 仅依赖 kmod-tun/luci-compat, 不自动拉 easytier 二进制包。
# 必须显式启用, 否则固件只有 LuCI/配置, 没有 /usr/bin/easytier-core。
# 注: easytier-core 二进制取决于 immortalwrt feeds 是否含 easytier 包;
#     v1 不强制注入, 若 feeds 缺包则此项为无害 no-op (详见 PRIVATE.sh 注释)。
CONFIG_PACKAGE_easytier=y

# ============ 裸核 sing-box ============
# sing-box 二进制由 PRIVATE.sh 从 reF1nd/sing-box-releases 注入到 /usr/bin/sing-box,
# 因此上游 feed 中的源码包不要编译。
CONFIG_PACKAGE_sing-box=n

# ============ reF1nd sing-box eBPF / TProxy 内核要求 ============
# beta.14 linux-arm64-musl is built with with_ebpf. shared_network uses TC
# and therefore requires cls_bpf + act_bpf from kmod-sched-bpf.
CONFIG_PACKAGE_kmod-sched-core=y
CONFIG_PACKAGE_kmod-sched-bpf=y
CONFIG_PACKAGE_kmod-nft-tproxy=y
CONFIG_PACKAGE_kmod-nft-socket=y
